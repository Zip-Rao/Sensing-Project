% Preamble for reconstruction_platform_report.pdf
% Used via: pandoc ... --include-in-header=_pandoc_preamble.tex

% --- Chinese support (xeCJK) ---
\usepackage{xeCJK}
\setCJKmainfont{Microsoft YaHei}
\setCJKmonofont{Microsoft YaHei}

% --- Code block: allow line breaks inside verbatim, smaller font ---
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  breaklines=true,
  breakanywhere=true,
  breakautoindent=true,
  fontsize=\small,
  commandchars=\\\{\}
}

% --- Inline code: smaller font (texttt break-on-anything handled by sloppy) ---
\let\oldtexttt\texttt
\renewcommand{\texttt}[1]{{\ttfamily\small #1}}

% --- Looser justification so long URL/path-in-code can wrap ---
\sloppy
\emergencystretch=3em

% --- Hyperlinks: allow line breaks; render as colored text ---
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=blue!60!black,
  urlcolor=blue!60!black,
  citecolor=blue!60!black,
  filecolor=blue!60!black,
  breaklinks=true,
  bookmarksopen=true,
  bookmarksdepth=3,
  pdftitle={Reconstruction Platform Technical Report},
  pdfauthor={sqc/ framework}
}

% --- Tables: allow paragraph wrapping inside cells ---
\usepackage{array}
\usepackage{longtable}
\renewcommand{\arraystretch}{1.15}

% --- Section spacing tighter to fit more on page ---
\usepackage{titlesec}
\titlespacing*{\section}{0pt}{1.5ex plus 0.5ex minus 0.2ex}{1ex plus 0.2ex}
\titlespacing*{\subsection}{0pt}{1.2ex plus 0.4ex minus 0.2ex}{0.8ex plus 0.2ex}
\titlespacing*{\subsubsection}{0pt}{1.0ex plus 0.3ex minus 0.2ex}{0.6ex plus 0.2ex}

% --- Math: enable common AMS packages (pandoc loads amsmath; we add amssymb) ---
\usepackage{amssymb}
\usepackage{mathtools}

% --- Map Unicode symbols missing from Times New Roman to LaTeX math equivalents ---
\usepackage{newunicodechar}
\newunicodechar{⇄}{\ensuremath{\rightleftarrows}}
\newunicodechar{∈}{\ensuremath{\in}}
\newunicodechar{⟨}{\ensuremath{\langle}}
\newunicodechar{⟩}{\ensuremath{\rangle}}

% --- Wider tables: shrink to text width if overflowing ---
\setlength{\tabcolsep}{4pt}

% --- Avoid Pandoc's "tightlist" undefined error ---
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
